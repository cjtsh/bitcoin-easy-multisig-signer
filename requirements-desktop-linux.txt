-r requirements.txt
# Linux needs no window toolkit. gui.py already serves the interface on
# localhost and opens the user's browser, so pywebview (and the WebKit/WebView2
# runtime it would drag in) is deliberately absent here. That is why this file
# is shorter than requirements-desktop.txt rather than a copy of it.
pyinstaller==6.22.2
# certifi ships the app's HTTPS CA trust store, so it is pinned to one exact
# version: an open range would silently adopt a future (or compromised) release.
# 2026.7.22 is the newest version satisfying the previous >=2025.1.31,<2027 range.
certifi==2026.7.22
hwi==3.2.0
# The Jade's PIN unlock is relayed through Blockstream's pin server. hwilib
# disables that relay unless `requests` is importable, and without it a Jade
# cannot be unlocked at all. The PIN itself never leaves the device.
requests==2.32.5
